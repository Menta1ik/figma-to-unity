using UnityEngine;
using TMPro;
using System.Linq;
using FigmaImporter.V2.Data;

namespace FigmaImporter.V2.Core.Handlers
{
    public class TextHandler : IFigmaComponentHandler
    {
        public bool CanHandle(FigmaNode node) => node.type == "TEXT";

        public void Apply(FigmaNode node, FigmaElement target, FigmaHandlerContext context)
        {
            var tmp = target.GetComponent<TextMeshProUGUI>();
            if (tmp == null) tmp = target.gameObject.AddComponent<TextMeshProUGUI>();
            
            tmp.text = node.characters;
            
            if (node.style != null)
            {
                tmp.fontSize = node.style.fontSize;
                tmp.alignment = MapAlignment(node.style.textAlignHorizontal, node.style.textAlignVertical);
            }
            
            if (node.fills != null)
            {
                var textFill = node.fills.FirstOrDefault(f => f.type == "SOLID");
                if (textFill != null && textFill.color != null) 
                {
                    tmp.color = textFill.color.ToUnityColor();
                }
            }
        }

        private TextAlignmentOptions MapAlignment(string h, string v)
        {
            if (h == "CENTER" && v == "CENTER") return TextAlignmentOptions.Center;
            if (h == "RIGHT") return TextAlignmentOptions.Right;
            return TextAlignmentOptions.Left;
        }
    }
}
using UnityEngine;
using UnityEngine.UI;
using System.Linq;
using FigmaImporter.V2.Data;

namespace FigmaImporter.V2.Core.Handlers
{
    public class ImageHandler : IFigmaComponentHandler
    {
        public bool CanHandle(FigmaNode node)
        {
            if (node.type == "TEXT") return false;
            // Обрабатываем узлы с заливкой (Solid, Image и т.д.)
            return node.fills != null && node.fills.Count > 0;
        }

        public void Apply(FigmaNode node, FigmaElement target, FigmaHandlerContext context)
        {
            var image = target.GetComponent<Image>();
            if (image == null) image = target.gameObject.AddComponent<Image>();

            // Если есть заливка типа IMAGE или градиент — ставим в очередь на скачивание
            bool hasImageFill = node.fills.Any(f => f.type == "IMAGE");
            bool hasGradientFill = node.fills.Any(f => f.type.StartsWith("GRADIENT"));
            
            // Если это сложный вектор, который мы не умеем рисовать uGUI примитивами
            bool isComplexVector = node.type == "VECTOR" || node.type == "STAR" || node.type == "POLYGON" || node.type == "ELLIPSE";

            if (hasImageFill || hasGradientFill || (isComplexVector && node.fills.Count > 0))
            {
                // Оставляем прозрачным до загрузки! Не делаем серым, иначе если скачивание упадет, останется уродливый квадрат.
                image.color = new Color(1f, 1f, 1f, 0f);
                context.ImageNodesToDownload.Add(node);
                return;
            }

            // Иначе, пытаемся применить сплошной цвет (SOLID)
            var solidFill = node.fills.FirstOrDefault(f => f.type == "SOLID" && f.visible != false);
            if (solidFill != null && solidFill.color != null) 
            {
                float combinedOpacity = node.opacity * solidFill.opacity;
                image.color = solidFill.color.ToUnityColor(combinedOpacity);
                // Отключаем скачивание спрайта, это просто цветной квадрат
                image.sprite = null; 
            }
            else
            {
                // Если заливка невидимая или нет заливки, делаем прозрачным
                image.color = new Color(0, 0, 0, 0);
            }
        }
    }
}
using UnityEngine;
using UnityEngine.UI;
using FigmaImporter.V2.Data;

namespace FigmaImporter.V2.Core.Handlers
{
    public class TransformHandler : IFigmaComponentHandler
    {
        public bool CanHandle(FigmaNode node) => node.absoluteBoundingBox != null;

        public void Apply(FigmaNode node, FigmaElement target, FigmaHandlerContext context)
        {
            RectTransform rt = target.GetComponent<RectTransform>();
            if (rt == null) return;

            rt.anchorMin = new Vector2(0, 1);
            rt.anchorMax = new Vector2(0, 1);
            rt.pivot = new Vector2(0, 1);

            target.AbsoluteBox = new Rect(node.absoluteBoundingBox.x, node.absoluteBoundingBox.y, 
                                        node.absoluteBoundingBox.width, node.absoluteBoundingBox.height);

            FigmaElement parentElement = target.transform.parent != null ? target.transform.parent.GetComponent<FigmaElement>() : null;
            
            float localX = 0; float localY = 0;
            if (parentElement != null)
            {
                localX = node.absoluteBoundingBox.x - parentElement.AbsoluteBox.x;
                localY = -(node.absoluteBoundingBox.y - parentElement.AbsoluteBox.y);
            }

            rt.anchoredPosition = new Vector2(localX, localY);
            rt.sizeDelta = new Vector2(node.absoluteBoundingBox.width, node.absoluteBoundingBox.height);

            target.gameObject.name = node.name;

            // Обработка Clip Content (Маска)
            if (node.clipsContent)
            {
                var mask = target.GetComponent<RectMask2D>();
                if (mask == null) target.gameObject.AddComponent<RectMask2D>();
            }
        }
    }
}
